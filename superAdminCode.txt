import { useState } from 'react';
import { useNavigate } from 'react-router-dom';
import { Eye, EyeOff } from 'lucide-react';

const campuses = ['Ikeja', 'Victoria Island', 'Lekki', 'Surulere', 'Yaba'];
const levels = ['Level 1', 'Level 2', 'Level 3', 'Level 4', 'Level 5'];

export default function AdminLogin() {
  const navigate = useNavigate();
  const [campus, setCampus] = useState('');
  const [level, setLevel] = useState('');
  const [password, setPassword] = useState('');
  const [showPassword, setShowPassword] = useState(false);

  function handleSignIn(e: React.FormEvent) {
    e.preventDefault();
    navigate('/admin/create');
  }

  return (
    <div className="min-h-screen w-full flex" style={{ fontFamily: 'Inter, sans-serif' }}>
      {/* Left panel — geometric background */}
      <div className="relative flex-[3] bg-[#f8fafc] overflow-hidden">
        {/* Geometric shapes — subtle overlapping circles and squares */}
        <div className="absolute inset-0 pointer-events-none">
          {/* Large circles */}
          <div className="absolute w-[420px] h-[420px] rounded-full border-[40px] border-[#e2e8f0] opacity-60"
               style={{ top: '-80px', left: '-100px' }} />
          <div className="absolute w-[300px] h-[300px] rounded-full border-[30px] border-[#e2e8f0] opacity-40"
               style={{ top: '120px', left: '260px' }} />
          <div className="absolute w-[500px] h-[500px] rounded-full border-[50px] border-[#f1f5f9] opacity-70"
               style={{ bottom: '-160px', left: '-60px' }} />
          <div className="absolute w-[200px] h-[200px] rounded-full border-[20px] border-[#e2e8f0] opacity-50"
               style={{ bottom: '80px', left: '340px' }} />
          {/* Squares / rectangles */}
          <div className="absolute w-[160px] h-[160px] rounded-[16px] border-[20px] border-[#e2e8f0] opacity-40"
               style={{ top: '200px', left: '60px', transform: 'rotate(18deg)' }} />
          <div className="absolute w-[100px] h-[100px] rounded-[12px] border-[14px] border-[#e2e8f0] opacity-30"
               style={{ top: '420px', left: '380px', transform: 'rotate(-12deg)' }} />
          <div className="absolute w-[220px] h-[220px] rounded-[20px] border-[28px] border-[#f1f5f9] opacity-60"
               style={{ bottom: '60px', right: '40px', transform: 'rotate(8deg)' }} />
          {/* Filled shapes */}
          <div className="absolute w-[80px] h-[80px] rounded-full bg-[#e2e8f0] opacity-30"
               style={{ top: '340px', left: '500px' }} />
          <div className="absolute w-[50px] h-[50px] rounded-[8px] bg-[#e2e8f0] opacity-25"
               style={{ top: '80px', left: '460px', transform: 'rotate(22deg)' }} />
          <div className="absolute w-[120px] h-[120px] rounded-full bg-[#f1f5f9] opacity-50"
               style={{ bottom: '200px', left: '180px' }} />
        </div>

        {/* Branding text bottom-left */}
        <div className="absolute bottom-[40px] left-[48px]">
          <p className="text-[13px] font-semibold text-[#94a3b8] tracking-widest uppercase">ClassroomQA</p>
          <p className="text-[12px] text-[#cbd5e1] mt-[4px]">Smarter classroom engagement</p>
        </div>
      </div>

      {/* Right panel — login card */}
      <div className="flex-[2] bg-white flex items-center justify-center border-l border-[#e5e7eb]">
        <div className="w-full max-w-[360px] px-[40px] py-[48px]">
          {/* Label */}
          <p className="text-[11px] font-semibold text-[#94a3b8] uppercase tracking-[0.12em] mb-[20px]">
            Campus Admin Portal
          </p>

          {/* Heading */}
          <h1 className="text-[28px] font-bold text-[#0f172a] leading-tight mb-[32px]">
            Sign in
          </h1>

          <form onSubmit={handleSignIn} className="flex flex-col gap-[16px]">
            {/* Campus dropdown */}
            <div className="flex flex-col gap-[6px]">
              <label className="text-[13px] font-medium text-[#475569]">Campus</label>
              <div className="relative">
                <select
                  value={campus}
                  onChange={e => setCampus(e.target.value)}
                  required
                  className="w-full appearance-none bg-[#f9fafb] border border-[#e5e7eb] rounded-[6px] px-[12px] py-[10px] text-[14px] text-[#0f172a] focus:outline-none focus:ring-2 focus:ring-[#2563eb] focus:border-transparent cursor-pointer"
                >
                  <option value="" disabled>Select campus…</option>
                  {campuses.map(c => (
                    <option key={c} value={c}>{c}</option>
                  ))}
                </select>
                <div className="pointer-events-none absolute right-[12px] top-1/2 -translate-y-1/2">
                  <svg width="14" height="14" viewBox="0 0 14 14" fill="none">
                    <path d="M3 5l4 4 4-4" stroke="#94a3b8" strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round"/>
                  </svg>
                </div>
              </div>
            </div>

            {/* Level dropdown */}
            <div className="flex flex-col gap-[6px]">
              <label className="text-[13px] font-medium text-[#475569]">Level</label>
              <div className="relative">
                <select
                  value={level}
                  onChange={e => setLevel(e.target.value)}
                  required
                  className="w-full appearance-none bg-[#f9fafb] border border-[#e5e7eb] rounded-[6px] px-[12px] py-[10px] text-[14px] text-[#0f172a] focus:outline-none focus:ring-2 focus:ring-[#2563eb] focus:border-transparent cursor-pointer"
                >
                  <option value="" disabled>Select level…</option>
                  {levels.map(l => (
                    <option key={l} value={l}>{l}</option>
                  ))}
                </select>
                <div className="pointer-events-none absolute right-[12px] top-1/2 -translate-y-1/2">
                  <svg width="14" height="14" viewBox="0 0 14 14" fill="none">
                    <path d="M3 5l4 4 4-4" stroke="#94a3b8" strokeWidth="1.5" strokeLinecap="round" strokeLinejoin="round"/>
                  </svg>
                </div>
              </div>
            </div>

            {/* Password input */}
            <div className="flex flex-col gap-[6px]">
              <label className="text-[13px] font-medium text-[#475569]">Password</label>
              <div className="relative">
                <input
                  type={showPassword ? 'text' : 'password'}
                  value={password}
                  onChange={e => setPassword(e.target.value)}
                  required
                  placeholder="Enter password"
                  className="w-full bg-[#f9fafb] border border-[#e5e7eb] rounded-[6px] px-[12px] py-[10px] pr-[40px] text-[14px] text-[#0f172a] placeholder:text-[#94a3b8] focus:outline-none focus:ring-2 focus:ring-[#2563eb] focus:border-transparent"
                />
                <button
                  type="button"
                  onClick={() => setShowPassword(prev => !prev)}
                  className="absolute right-[10px] top-1/2 -translate-y-1/2 text-[#94a3b8] hover:text-[#475569] transition-colors"
                  tabIndex={-1}
                >
                  {showPassword ? <EyeOff size={16} /> : <Eye size={16} />}
                </button>
              </div>
            </div>

            {/* Sign in button */}
            <button
              type="submit"
              className="mt-[8px] w-full bg-[#2563eb] hover:bg-[#1d4ed8] text-white font-semibold text-[15px] py-[11px] rounded-[6px] transition-colors duration-150"
            >
              Sign in
            </button>
          </form>
        </div>
      </div>
    </div>
  );
}
